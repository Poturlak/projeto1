Código de Cores do Overlay
python
def calcular_cor_valor(valor_medido, referencia):
    if not referencia:
        return "cinza"      # Sem dados de referência
    
    media = referencia.media
    desvio = referencia.desvio_padrao
    
    if abs(valor_medido - media) > 3 * desvio:
        return "vermelho"   # Crítico - valor muito fora
    elif abs(valor_medido - media) > 2 * desvio:
        return "laranja"    # Atenção - possível problema
    elif abs(valor_medido - media) > desvio:
        return "amarelo"    # Leve variação - monitorar
    else:
        return "verde"      # Normal - dentro do esperado

CREATE TABLE usuarios (
    id VARCHAR PRIMARY KEY,
    nome VARCHAR,
    reputacao FLOAT,
    data_cadastro DATETIME
);

-- Metadados das placas
CREATE TABLE placas (
    id INTEGER PRIMARY KEY,
    metadata_json TEXT,
    imagem BLOB,
    usuario_id VARCHAR,
    publica BOOLEAN,
    data_criacao DATETIME
);

-- Pontos de medição
CREATE TABLE pontos (
    id INTEGER PRIMARY KEY,
    placa_id INTEGER,
    coordenadas_x INTEGER,
    coordenadas_y INTEGER,
    descricao VARCHAR,
    sequencia INTEGER
);

-- Medições realizadas
CREATE TABLE medicoes (
    id INTEGER PRIMARY KEY,
    ponto_id INTEGER,
    diodo FLOAT,
    resistencia FLOAT,
    tensao FLOAT,
    usuario_id VARCHAR,
    data_medicao DATETIME
);
